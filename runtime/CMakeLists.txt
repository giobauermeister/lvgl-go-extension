project(lvgl_go C CXX)

  message(FATAL_ERROR "Use emcmake cmake to configure the LVGL Go runtime")
endif()
if(NOT UI_DIR OR NOT EXISTS "${UI_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set UI_DIR to an exported LVGL Pro C project")
endif()

set(CMAKE_CXX_STANDARD 17)

  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Read project metadata for the display defaults.
file(READ "${UI_DIR}/project.xml" project_metadata)
string(REGEX MATCH "lvgl_version=\"([^\"]+)\"" _ "${project_metadata}")
set(default_lvgl_version "${CMAKE_MATCH_1}")
if(NOT default_lvgl_version)
  set(default_lvgl_version "9.5.0")
endif()
string(REGEX REPLACE "^v" "" default_lvgl_version "${default_lvgl_version}")
set(LVGL_VERSION "v${default_lvgl_version}" CACHE STRING "LVGL tag matching the exported project")
set(LVGL_SOURCE_DIR "" CACHE PATH "Explicit matching upstream LVGL source checkout")
set(LVGL_LIBRARY_DIR "" CACHE PATH "LVGL Pro versioned libs directory")

# Applies to custom object libraries and generated component libraries too.
add_compile_options("-sUSE_SDL=2")
add_compile_options("-ffile-prefix-map=${UI_DIR}=ui" "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=lvgl-go-runtime")
add_compile_definitions(LV_LVGL_H_INCLUDE_SIMPLE LV_SIM_BUILD LV_USE_XML=0)
if(LVGL_SOURCE_DIR)
  add_compile_options("-ffile-prefix-map=${LVGL_SOURCE_DIR}=lvgl")
  message(STATUS "Using local LVGL sources: ${LVGL_SOURCE_DIR}")
  set(LVGL_GO_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.defaults" CACHE FILEPATH "LVGL config defaults for this UI")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${LVGL_SOURCE_DIR}/scripts/generate_lv_conf.py"
      --template "${LVGL_SOURCE_DIR}/lv_conf_template.h"
      --defaults "${LVGL_GO_DEFAULTS}" --config "${CMAKE_BINARY_DIR}/lv_conf.h.in"
    RESULT_VARIABLE config_result OUTPUT_VARIABLE config_output ERROR_VARIABLE config_error)
  if(NOT config_result EQUAL 0)
    message(FATAL_ERROR "Generating LVGL config failed: ${config_output}\n${config_error}")
  endif()
  configure_file("${CMAKE_BINARY_DIR}/lv_conf.h.in" "${CMAKE_BINARY_DIR}/lv_conf.h" COPYONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LVGL_GO_DEFAULTS}")
  set(LV_BUILD_CONF_PATH "${CMAKE_BINARY_DIR}/lv_conf.h" CACHE PATH "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/lvgl")
elseif(LVGL_LIBRARY_DIR)
  message(STATUS "Using LVGL Pro precompiled ${LVGL_VERSION}: ${LVGL_LIBRARY_DIR}/lib/liblvgl.a")
  set(lvgl_config "${LVGL_LIBRARY_DIR}/include/lvgl/config/lv_conf.h")
  if(NOT EXISTS "${lvgl_config}")
    set(lvgl_config "${LVGL_LIBRARY_DIR}/include/lvgl/lv_conf.h")
  endif()
  foreach(required_file lib/liblvgl.a include/lvgl/lvgl.h)
    if(NOT EXISTS "${LVGL_LIBRARY_DIR}/${required_file}")
      message(FATAL_ERROR "LVGL Pro library is incomplete: ${LVGL_LIBRARY_DIR}/${required_file}")
    endif()
  endforeach()
  if(NOT EXISTS "${lvgl_config}")
    message(FATAL_ERROR "LVGL Pro library is missing lv_conf.h in include/lvgl/config or include/lvgl")
  endif()
  # Headers must use exactly the configuration used to build the archive.
  # Import LVGL and its separately packaged ThorVG dependency when supplied.
  add_compile_definitions(LV_CONF_PATH="${lvgl_config}")
  set(lvgl_include_dirs "${LVGL_LIBRARY_DIR}/include" "${LVGL_LIBRARY_DIR}/include/lvgl")
  # Custom widgets include lvgl_private.h directly in simple-include mode.
  # Newer Pro SDKs package it separately from the public headers.
  if(EXISTS "${LVGL_LIBRARY_DIR}/include/lvgl_private/lvgl_private.h")
    list(APPEND lvgl_include_dirs "${LVGL_LIBRARY_DIR}/include/lvgl_private")
  endif()
  include_directories(${lvgl_include_dirs})
  set(LVGL_GO_PREBUILT ON)
  add_library(lvgl STATIC IMPORTED GLOBAL)
  set_target_properties(lvgl PROPERTIES
    IMPORTED_LOCATION "${LVGL_LIBRARY_DIR}/lib/liblvgl.a"
    INTERFACE_INCLUDE_DIRECTORIES "${lvgl_include_dirs}")
  if(EXISTS "${LVGL_LIBRARY_DIR}/lib/liblvgl_thorvg.a")
    add_library(lvgl_thorvg STATIC IMPORTED GLOBAL)
    set_target_properties(lvgl_thorvg PROPERTIES
      IMPORTED_LOCATION "${LVGL_LIBRARY_DIR}/lib/liblvgl_thorvg.a")
    target_link_libraries(lvgl INTERFACE lvgl_thorvg)
  endif()
else()
  message(FATAL_ERROR "Matching LVGL Pro WebAssembly library not found. Open this project in LVGL Pro to install its versioned resources, or set runtime.lvglLibraryDir / runtime.lvglSourceDir.")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${UI_DIR}/project.xml")
add_subdirectory("${UI_DIR}" "${CMAKE_BINARY_DIR}/ui")

# Build the C export with the sources used by the browser preview.
function(lvgl_go_remove_xml_parsers directory)
  get_property(ui_targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(ui_target IN LISTS ui_targets)
    get_target_property(ui_sources "${ui_target}" SOURCES)
    if(ui_sources)
      list(FILTER ui_sources EXCLUDE REGEX "_xml_parser\\.c$")
      set_property(TARGET "${ui_target}" PROPERTY SOURCES "${ui_sources}")
    endif()
  endforeach()
  get_property(ui_subdirectories DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(ui_directory IN LISTS ui_subdirectories)
    lvgl_go_remove_xml_parsers("${ui_directory}")
  endforeach()
endfunction()
lvgl_go_remove_xml_parsers("${UI_DIR}")
